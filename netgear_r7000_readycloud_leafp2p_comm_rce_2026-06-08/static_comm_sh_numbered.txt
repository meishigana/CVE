    23	# arg: <data> [<store.path>]
    24	#
    25	comm_post()
    26	{
    27	    local post="${TMP_PREFIX}/readycloud_r.post"
    28	    echo "${1}" > "${post}"
    29	    comm_post_file "${post}" "${2}" || {
    30	# 	rm -f "${post}"
    31		return $ERROR
    32	    }
    33	#     rm -f "${post}"
    34	    return $OK
    35	}
    36	
    37	#
    38	# args: <file.path> [<store.path>]
    39	#
    40	comm_post_file()
    41	{
    42	    COMM_RESULT=""
    43	    [ -z "${1}" ] && return $ERROR
    44	    if [ -z "${2}" ];
    45	    then
    46		FULL_EXEC="\`cat "${1}" | ${COMM_EXEC} -X POST --data-binary @- 2>/dev/null\`"
    47	# 	FULL_EXEC="\`cat "${1}" | ${COMM_EXEC} -X POST --data-binary @- \`"
    48	    else
    49	# 	FULL_EXEC="\`cat "${1}" | ${COMM_EXEC} -X POST --data-binary @- > "${2}"\`"
    50		FULL_EXEC="\`cat "${1}" | ${COMM_EXEC} -X POST --data-binary @- 2>/dev/null -o '${2}'\`"
    51	    fi
    52	#     echo "${FULL_EXEC}"
    53	    eval COMM_RESULT="${FULL_EXEC}" || return $ERROR
    54	
    55	    return $OK
    56	}

    58	#
    59	# args: <user name> <password>
    60	#
    61	do_register()
    62	{
    63		# construct request
    64		USER_NAME=$1
    65		USER_PASS=$2
    66		temp_dir=$3
    67		XAGENT_ID=$(readycloud_nvram get x_agent_id | tr -d '[[:space:]]')
    68		MODEL=$(remote_smb_conf -get_model_name | tr -d '[[:space:]]')
    69		SERIAL_NUMBER=$(system get serial_number | tr -d '[[:space:]]')
    70	
    71		FIRMWARE_VERSION=`version | sed -n 2p | awk -F "/" '{print $2}' | sed -r 's/^.{1}//'`
    72		#get second line of "version" command output
    73		#get second part of "U12H270T00/V1.0.3.49/20140403_xAgent" line (version)
    74		#and removing first character "V" from it
    75		#output - "1.0.3.49"
    76	
    77		DATA="<?xml version=\"1.0\" encoding=\"utf-8\"?>"
    78		DATA="${DATA}<request moniker=\"/root/devices\" method=\"register\">"
    79		DATA="${DATA}<body type=\"registration\">"
    80		DATA="${DATA}<username>${USER_NAME}</username>"
    81		DATA="${DATA}<password><![CDATA[${USER_PASS}]]></password>"
    82		DATA="${DATA}<model>${MODEL}</model>"
    83		DATA="${DATA}<firmware_id>${FIRMWARE_VERSION}</firmware_id>"
    84	    DATA="${DATA}<x_agent_id>${XAGENT_ID}</x_agent_id>"
    85		DATA="${DATA}<license><LicenseKey>sdfsfgjsflkj</LicenseKey><hardwareSN>${SERIAL_NUMBER}</hardwareSN><StartTime>0</StartTime><ExpiredTime>999</ExpiredTime><valid>true</valid></license>"
    86		DATA="${DATA}</body></request>"
    87	
    88		comm_post "${DATA}" && {
    89		   if [ "xSUCCESS" = "x${COMM_RESULT}" ]; then
    90			readycloud_nvram set readycloud_registration_owner=${USER_NAME}
    91	        readycloud_nvram set leafp2p_run="1"
    92	        readycloud_nvram set x_force_connection="1"
    93	        readycloud_nvram commit
    94	        kill -SIGHUP `cat /tmp/xagent_watchdog.pid`
    95	
    96			internet set connection readycloud 1
    97			return $OK
    98		    fi
    99		}
   100		echo "Invalid User Name or Password"
   101		return $ERROR
   102	}
